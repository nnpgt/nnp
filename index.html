sysctl -a | grep ip_forward
apt-get update && apt-get install -y iptables
iptables -t nat -A POSTROUTING -s 172.16.1.0/28 -o ens20 -j MASQUERADE
iptables -t nat -A POSTROUTING -s 172.16.2.0/28 -o ens20 -j MASQUERADE
iptables-save >> /etc/sysconfig/iptables
systemctl enable --now iptables
iptables -t nat -L -n -v

5Безопасный удалённый доступ (SSH)

HQ-SRV и BR-SRV

sed -i -E '/^\s*#?\s*(Port|AllowUsers|MaxAuthTries|Banner)\s/d' /etc/openssh/sshd_config
cat >> /etc/openssh/sshd_config <<'EOF'
Port 2026
AllowUsers sshuser
MaxAuthTries 2
Banner /etc/openssh/banner
EOF
echo "Authorized access only" > /etc/openssh/banner
systemctl restart sshd
# проверка
ssh sshuser@localhost
ssh -p 2026 sshuser@localhost
ssh -p 2026 user@localhost

10DNS-сервер на HQ-SRV

HQ-SRV · установка и options.conf

apt-get update && apt-get install bind bind-utils -y
sed -i -E 's|^\s*(//)?\s*listen-on\s*\{.*|listen-on { 192.168.100.2; };|' /var/lib/bind/etc/options.conf
sed -i -E 's|^\s*(//)?\s*listen-on-v6\s*\{.*|listen-on-v6 { none; };|' /var/lib/bind/etc/options.conf
sed -i -E 's|^\s*(//)?\s*forwarders\s*\{.*|forwarders { 77.88.8.8; };|' /var/lib/bind/etc/options.conf
sed -i -E 's|^\s*(//)?\s*allow-query\s*\{.*|allow-query { any; };|' /var/lib/bind/etc/options.conf

HQ-SRV · зоны

cat >> /var/lib/bind/etc/rfc1912.conf <<'EOF'
zone "au-team.irpo" {
    type master;
    file "au-team.irpo";
};
zone "100.168.192.in-addr.arpa" {
    type master;
    file "100.168.192.in-addr.arpa";
};
zone "200.168.192.in-addr.arpa" {
    type master;
    file "200.168.192.in-addr.arpa";
};
EOF
cp /var/lib/bind/etc/zone/empty /var/lib/bind/etc/zone/au-team.irpo
cp /var/lib/bind/etc/zone/empty /var/lib/bind/etc/zone/100.168.192.in-addr.arpa
cp /var/lib/bind/etc/zone/empty /var/lib/bind/etc/zone/200.168.192.in-addr.arpa
cat > /var/lib/bind/etc/zone/au-team.irpo <<'EOF'
$TTL 1D
@   IN  SOA au-team.irpo. root.au-team.irpo. (
            2025062300 ; serial
            12H        ; refresh
            1H         ; retry
            1W         ; expire
            1H         ; ncache
)
    IN  NS  au-team.irpo.
    IN  A   192.168.100.2
hq-srv  IN  A   192.168.100.2
hq-cli  IN  A   192.168.200.2
hq-rtr  IN  A   192.168.100.1
hq-rtr  IN  A   192.168.200.1
hq-rtr  IN  A   192.168.99.1
docker  IN  A   172.16.1.1
web     IN  A   172.16.2.1
br-srv  IN  A   192.168.0.2
br-rtr  IN  A   192.168.0.1
EOF
cat > /var/lib/bind/etc/zone/100.168.192.in-addr.arpa <<'EOF'
$TTL 1D
@   IN  SOA au-team.irpo. root.au-team.irpo. (
            2025062300 ; serial
            12H        ; refresh
            1H         ; retry
            1W         ; expire
            1H         ; ncache
)
    IN  NS  au-team.irpo.
1   IN  PTR hq-rtr.au-team.irpo.
2   IN  PTR hq-srv.au-team.irpo.
EOF
cat > /var/lib/bind/etc/zone/200.168.192.in-addr.arpa <<'EOF'
$TTL 1D
@   IN  SOA au-team.irpo. root.au-team.irpo. (
            2025062300 ; serial
            12H        ; refresh
            1H         ; retry
            1W         ; expire
            1H         ; ncache
)
    IN  NS  au-team.irpo.
1   IN  PTR hq-rtr.au-team.irpo.
2   IN  PTR hq-cli.au-team.irpo.
EOF
